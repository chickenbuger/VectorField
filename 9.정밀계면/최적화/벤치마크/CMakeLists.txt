cmake_minimum_required(VERSION 3.20)
project(Stage9PerformanceEvidence LANGUAGES CXX)
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
execute_process(COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/PrepareBaseline.ps1" -RepositoryRoot "${REPOSITORY_ROOT}" -OutputDirectory "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE PREPARE_RESULT)
if(NOT PREPARE_RESULT EQUAL 0)
  message(FATAL_ERROR "Preparing baseline failed")
endif()
add_executable(baseline_profile BaselineProfile.cpp)
target_compile_features(baseline_profile PRIVATE cxx_std_14)
target_include_directories(baseline_profile PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${REPOSITORY_ROOT}/opgnel_project_smoke")
add_executable(compare_interface_performance ComparePerformance.cpp)
target_compile_features(compare_interface_performance PRIVATE cxx_std_14)
target_include_directories(compare_interface_performance PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${REPOSITORY_ROOT}/opgnel_project_smoke")
if(MSVC)
  target_compile_options(baseline_profile PRIVATE /O2 /fp:precise)
  target_compile_options(compare_interface_performance PRIVATE /O2 /fp:precise)
endif()
